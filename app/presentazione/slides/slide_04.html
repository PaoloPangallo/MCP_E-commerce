def _update_budget(user, category, signal):
    if not signal: return
    
    cat_key = f"auto_cat:{category.lower()}"
    if cat_key in budgets:
        old = float(budgets[cat_key])
        # EMA calculation α=0.20
        new_val = old * 0.80 + signal * 0.20
    else:
        new_val = signal  # Cold start
        
    budgets[cat_key] = round(new_val, 2)
